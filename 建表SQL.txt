-- ============================================================
--  反诈先锋 · 留言墙
-- ============================================================

create table if not exists public.messages (
  id          bigserial primary key,
  created_at  timestamptz   not null default now(),
  nickname    text          not null default '',
  content     text          not null,
  score       int,
  identity    text,
  client_fp   text          not null default '',
  is_hidden   boolean       not null default false
);

-- 常用查询：只取未隐藏的、按时间倒序
create index if not exists messages_visible_idx
  on public.messages (created_at desc)
  where is_hidden = false;

-- 指纹限流用
create index if not exists messages_fp_idx
  on public.messages (client_fp, created_at desc);

-- ============================================================
--  访问控制：两层，缺一不可
--  RLS 决定「哪些行能碰」，grant 决定「哪些操作能做」。
--  少了任何一层，数据库就是对全互联网开放的。
-- ============================================================
alter table public.messages enable row level security;

-- 先收干净：anon 什么都不能做
revoke all on public.messages from anon, authenticated;

-- 只给两个最小权限：读、写。**不 grant delete、不 grant update**。
-- 没有 delete 权限 → 前端即使被改也删不掉任何数据。
-- 没有 update 权限 → 「先发一条正常留言、过两天改成广告」这条路也堵死。
grant select, insert on public.messages to anon;

-- 登录用户（将来你若做后台登录）也只给读
grant select on public.messages to authenticated;

-- 序列（bigserial 的 id 需要它，否则 insert 会报 permission denied）
grant usage, select on sequence public.messages_id_seq to anon;


-- ---------- 策略 1：所有人都能看未隐藏的留言 ----------
drop policy if exists "public can read visible" on public.messages;
create policy "public can read visible"
  on public.messages for select
  to anon, authenticated
  using (is_hidden = false);
  -- ⚠️ 注意这里**没有** is_hidden=true 的例外：被隐藏的留言连自己都查不到，
  --    前端拿不到 = 真正生效。隐藏后要复查就去 Supabase 的 Table Editor 看。

-- ---------- 策略 2：任何人都能插入，但字段必须合规 ----------
-- 刻意**不建** delete 策略、**不建** update 策略、**不用** for all。
drop policy if exists "anon can insert" on public.messages;
create policy "anon can insert"
  on public.messages for insert
  to anon
  with check (
    is_hidden = false                                   -- 不能自己把留言设成隐藏
    and char_length(content) between 1 and 200
    and char_length(nickname) <= 12
    and (score is null or (score >= 0 and score <= 100))
    and (identity is null or identity in ('youth','mid','elder'))
    and client_fp <> ''
  );


-- ============================================================
--  发布走 RPC：限流必须在数据库侧
--  前端那个 30 秒冷却改一下 localStorage 就绕过了，只有这里算数。
-- ============================================================
create or replace function public.post_message(
  nickname  text,
  content   text,
  score     int,
  identity  text,
  client_fp text
) returns void
language plpgsql
security definer            -- 以函数属主身份插入，anon 无需直接插表权限
set search_path = public
as $$
declare
  recent    int;
  total_day int;
begin
  if char_length(coalesce(content,'')) < 1 or char_length(content) > 200 then
    raise exception '内容长度必须是 1~200 字' using errcode = '22023';
  end if;
  if length(coalesce(client_fp,'')) < 4 then
    raise exception '缺少客户端标识' using errcode = '22023';
  end if;

  -- 同一个指纹 10 秒内最多 1 条
  -- ⚠️ 必须用表别名 m 明确限定列：函数参数和表列都叫 client_fp，
  --    PL/pgSQL 默认 variable_conflict=error 时会直接抛
  --    42702「column reference "client_fp" is ambiguous」，
  --    后果是**任何合法留言都写不进去**。而且很能骗人 ——
  --    空内容那条校验在它之前就 raise 了，看起来「服务端校验正常」。
  select count(*) into recent from public.messages m
   where m.client_fp = post_message.client_fp
     and m.created_at > now() - interval '10 seconds';
  if recent >= 1 then
    raise exception '发得太快了，10 秒后再试' using errcode = 'P0001';
  end if;

  -- 同一个指纹一天最多 5 条（同样必须限定列，理由见上）
  select count(*) into total_day from public.messages m
   where m.client_fp = post_message.client_fp
     and m.created_at > now() - interval '1 day';
  if total_day >= 5 then
    raise exception '今天留言有点多，明天再来' using errcode = 'P0001';
  end if;

  -- values 里的形参一律加 post_message. 前缀，彻底消除与列名同名的歧义
  insert into public.messages (nickname, content, score, identity, client_fp)
  values (
    left(coalesce(nullif(trim(post_message.nickname),''), ''), 12),
    left(trim(post_message.content), 200),
    post_message.score, post_message.identity, post_message.client_fp
  );
end;
$$;

-- 只允许 anon 调这个函数（不给 authenticated，将来做后台再说）
revoke all on function public.post_message(text,text,int,text,text) from public;
grant execute on function public.post_message(text,text,int,text,text) to anon;
